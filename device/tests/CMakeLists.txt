cmake_minimum_required(VERSION 3.16)

add_library(eh_unity STATIC vendor/unity/unity.c)
target_include_directories(eh_unity PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/vendor/unity)

function(eh_add_test name)
  set(target eh_${name})
  add_executable(${target} ${name}.c)
  target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${EH_ROOT}/include)
  target_link_libraries(${target} PRIVATE edgehub_device eh_unity)
  if(NOT WIN32)
    target_link_libraries(${target} PRIVATE m)
  endif()
  if(MSVC)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  add_test(NAME ${target} COMMAND ${target})
endfunction()

eh_add_test(test_values_writer)
eh_add_test(test_hmac_golden)
eh_add_test(test_checkin)
eh_add_test(test_telemetry)
eh_add_test(test_commands)
eh_add_test(test_scheduling)
